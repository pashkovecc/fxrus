country_event = {
	id = sovietscorched.1
	immediate = {log = "[GetDateText]: [Root.GetName]: event sovietscorched.1"}
	title = sovietscorched.1.t
	desc = sovietscorched.1.d
	picture = soviet_farmers

	is_triggered_only = yes

	option = {
		name = sovietscorched.1.a
		every_state = {
			limit = {
				has_state_flag = SOV_scorched_ukraine
			}
			THIS = {
				clr_state_flag = SOV_scorched_ukraine
				add_state_modifier = {
					modifier = {
						local_org_regain = 0.5
						local_intel_to_enemies = 0.5
						local_supplies = 0.5
						local_building_slots_factor = 0.25
						local_non_core_manpower = 1.0
						local_manpower = 0.5
						local_resources_factor = 0.5
						local_resources = 0.5
					}
				}
			}
		}
	}
}
